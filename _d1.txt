232|     def geocode_place(self, place: str) -> Optional[Location]:
233|         """Ubah nama daerah menjadi Location (bbox + area id).
234| 
235|         Return None bila daerah tidak ditemukan atau request gagal.
236|         """
237|         cleaned = _WHITESPACE_RE.sub(" ", str(place or "")).strip()
238|         if not cleaned:
239|             return None
240| 
241|         params = {
242|             "q": cleaned,
243|             "format": "jsonv2",
244|             "limit": 1,
245|             "addressdetails": 1,
246|         }
247| 
248|         try:
249|             response = self.session.get(
250|                 self.nominatim_url, params=params, timeout=NOMINATIM_TIMEOUT
251|             )
252|         except requests.exceptions.RequestException as exc:
253|             self.last_errors.append(f"Nominatim gagal dihubungi: {self._short(exc)}")
254|             return None
255| 
256|         if response.status_code != 200:
257|             self.last_errors.append(
258|                 f"Nominatim HTTP {response.status_code} untuk daerah '{cleaned}'."
259|             )
260|             return None
261| 
262|         try:
263|             payload = response.json()
264|         except ValueError:
265|             self.last_errors.append("Respons Nominatim bukan JSON yang valid.")
266|             return None
267| 
268|         if not isinstance(payload, list) or not payload:
269|             self.last_errors.append(
270|                 f"Daerah '{cleaned}' tidak ditemukan di OpenStreetMap."
271|             )
272|             return None
273| 
274|         location = self._location_from_payload(payload[0], cleaned)
275|         if location is None:
276|             self.last_errors.append(
277|                 f"Daerah '{cleaned}' tidak punya bounding box yang bisa dipakai."
278|             )
279|         return location

281|     @staticmethod
282|     def _location_from_payload(item: Dict[str, Any], query: str) -> Optional[Location]:
283|         """Bangun Location dari satu objek hasil Nominatim."""
284|         try:
285|             south, north, west, east = (float(value) for value in (item.get("boundingbox") or [])[:4])
286|         except (TypeError, ValueError):
287|             return None
288| 
289|         address = item.get("address") or {}
290|         city = (
291|             address.get("city_district")
292|             or address.get("city")
293|             or address.get("town")
294|             or address.get("municipality")
295|             or address.get("county")
296|             or item.get("name")
297|             or query
298|         )
299| 
300|         try:
301|             osm_id = int(item.get("osm_id") or 0)
302|         except (TypeError, ValueError):
303|             osm_id = 0
304| 
305|         return Location(
306|             query=query,
307|             display_name=str(item.get("display_name") or ""),
308|             osm_type=str(item.get("osm_type") or ""),
309|             osm_id=osm_id,
310|             boundingbox=(south, north, west, east),
311|             city=title_case_place(str(city)),
312|             addresstype=str(item.get("addresstype") or ""),
313|             place_type=str(item.get("type") or ""),
314|         )

319|     @staticmethod
320|     def build_overpass_query(
321|         filters: Iterable[str],
322|         location: Location,
323|         limit: int = 50,
324|         timeout: int = OVERPASS_TIMEOUT,
325|     ) -> str:
326|         """Susun QL Overpass dari filter + Location.
327| 
328|         Bila geocode berupa relation administratif, pencarian dibatasi dengan
329|         ``area(3600000000 + osm_id)->.a``; bila tidak, dipakai bounding box.
330|         ``out center tags`` mengambil tag lengkap tanpa geometri (respons ringan).
331| 
332|         Raise ValueError bila location kosong (query global terlalu berat).
333|         """
334|         if location is None:
335|             raise ValueError("Location wajib diisi untuk menyusun query Overpass.")
336| 
337|         valid = [str(item).strip() for item in filters if str(item or "").strip()]
338|         if not valid:
339|             raise ValueError("Minimal satu filter Overpass harus diberikan.")
340| 
341|         header = f"[out:json][timeout={int(timeout)}];"
342|         area_id = location.area_id
343|         if area_id is not None:
344|             header += f"area({area_id})->.a;"
345|             scope = "(area.a)"
346|         else:
347|             scope = f"({location.bbox})"
348| 
349|         statements = "".join(f"nwr{filter_text}{scope};" for filter_text in valid)
350|         return f"{header}({statements});out center tags {int(limit)};"
